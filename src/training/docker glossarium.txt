# DOCKER
# ═════════════════════════════════════════════════════════════════════════════════════════════════════

# 1. MENJALANKAN
docker compose up                                    # jalankan (log tampil di terminal)
docker compose up -d                                 # jalankan di latar belakang
docker compose up --build                            # jalankan sambil membangun ulang image
docker compose up -d --build                         # (baru) latar belakang + bangun ulang image
docker compose up -d --force-recreate                # (baru) paksa buat ulang container tanpa build
docker compose up -d app                             # (baru) jalankan satu layanan saja
docker compose start                                 # nyalakan lagi container yang dihentikan

# 2. MENGHENTIKAN
docker compose stop                                  # hentikan container (tidak dihapus)
docker compose stop app                              # (baru) hentikan satu layanan saja
docker compose restart                               # nyalakan ulang tanpa membangun ulang
docker compose restart app                           # (baru) nyalakan ulang satu layanan saja
docker compose down                                  # hentikan dan hapus container + jaringan
docker compose down -v                               # sama, plus hapus volume bernama (hati-hati)
docker compose down --rmi all                        # (baru) sama, plus hapus image (build ulang dari nol)
docker compose down --remove-orphans                 # (baru) sama, plus hapus container sisa layanan lama

# 3. MELIHAT KONDISI
docker compose ps                                    # container milik projek ini saja
docker compose ps -a                                 # (baru) termasuk container projek yang sudah berhenti
docker compose top                                   # (baru) proses yang berjalan di dalam container
docker ps                                            # semua container yang sedang menyala
docker ps -a                                         # semua container, termasuk yang sudah berhenti
docker compose logs -f                               # ikuti log secara langsung (Ctrl + C untuk keluar)
docker compose logs app                              # log layanan tertentu (sekali tampil)
docker compose logs --tail 100 app                   # (baru) 100 baris log terakhir saja
docker compose logs -f --tail 50 app                 # (baru) ikuti log, mulai dari 50 baris terakhir
docker stats                                         # pemakaian CPU dan RAM secara langsung
docker images                                        # daftar image yang tersimpan
docker system df                                     # pemakaian disk oleh Docker
docker inspect nama-container                        # (baru) detail lengkap container (IP, mount, env, dll.)

# 4. MEMBANGUN ULANG
docker compose build                                 # bangun image tanpa menjalankan
docker compose build --no-cache                      # bangun ulang dari nol tanpa cache
docker compose build app                             # (baru) bangun satu layanan saja
docker compose pull                                  # (baru) unduh image terbaru (untuk layanan yang pakai image jadi)

# 5. MASUK & JALANKAN PERINTAH DI CONTAINER (container harus sedang menyala)
docker compose exec app sh                           # masuk ke terminal container (keluar: exit)
docker compose exec app npm run build                # build untuk produksi
docker compose exec app npm run lint                 # periksa kode dengan ESLint
docker compose exec app npm run dev                  # (baru) jalankan dev server manual di container
docker compose exec app npm ls                       # (baru) lihat daftar library terpasang
docker compose exec app npm outdated                 # (baru) cek library yang punya versi baru
docker compose run --rm app sh                       # (baru) masuk terminal container sekali pakai (container belum perlu menyala)

# 6. KELOLA LIBRARY (perlu koneksi internet / cek jaringan)
docker compose exec app npm install lucide-react     # pasang library
docker compose exec app npm install -D eslint        # (baru) pasang sebagai devDependency
docker compose exec app npm uninstall lucide-react   # hapus library
docker compose exec app npm update                   # (baru) perbarui library sesuai rentang versi
docker compose run --rm app npm install lucide-react # alternatif kalau container belum menyala

# 7. VOLUME & JARINGAN  (baru)
docker volume ls                                     # daftar semua volume
docker volume inspect nama-volume                    # detail satu volume
docker volume rm nama-volume                         # hapus volume tertentu (misal untuk reset node_modules)
docker volume prune                                  # hapus semua volume yang tidak dipakai (hati-hati)
docker network ls                                    # daftar jaringan Docker
docker network prune                                 # hapus jaringan yang tidak dipakai

# 8. SALIN FILE  (baru)
docker cp nama-container:/app/file.txt ./file.txt    # salin dari container ke host
docker cp ./file.txt nama-container:/app/file.txt    # salin dari host ke container

# 9. PENGECEKAN KONFIGURASI  (baru)
docker compose config                                # validasi dan tampilkan compose.yml hasil akhir
docker compose config --services                     # daftar nama layanan
docker version                                       # versi Docker client dan server
docker compose version                               # versi Docker Compose
docker info                                          # info sistem Docker

# 10. MEMBERSIHKAN
docker rm -f nama-container                          # hapus container tertentu
docker rmi nama-image                                # hapus image tertentu
docker container prune                               # hapus semua container yang berhenti
docker image prune                                   # hapus image tidak terpakai (dangling)
docker builder prune                                 # bersihkan cache build
docker system prune                                  # bersihkan sisa container, jaringan, dan image dangling
docker system prune -a                               # lebih agresif, termasuk image yang tidak dipakai container mana pun
docker system prune -a --volumes                     # (baru) paling agresif, volume ikut terhapus (sangat hati-hati)


docker compose up -d      # mulai kerja
docker compose stop       # selesai kerja

* docker compose exec app npm ... memakai nama layanan app (sesuai docker-compose.yml), sedangkan docker rm, docker cp, dan docker inspect memakai nama container, yaitu habit-tracker-app

* Kalau nanti menambah library lalu down & jalankan ini ketika menambah library atau mengubah package.json/Dockerfile : lakukan docker compose down -v && docker compose up -d --build agar library masuk ke image.
